# MongoDB Collections Overview — Database: `ytscan`

> This document describes all five main collections in the **ytscan** MongoDB database.  
> It reflects the **v7.x schema (local mode)** — the current structure used in `yt-autoscanner` with ML-ready processing, dashboard exports, and automatic tracking logic.

---

## 1️⃣ Collection: `videos` (CORE)

**Purpose:** Raw video-level data collected by `discover_once.py` and updated by `track_once.py`.

### Structure
```json
{
  "_id": "VIDEO_ID",
  "ml_flags": {
    "likely_viral": false,
    "viral_confirmed": false,
    "score": 0.0
  },
  "source": {
    "query": "travel vlog japan",
    "regionCode": "JP",
    "randomMode": false,
    "channelHandle": "@travelerA"
  },
  "snippet": {
    "title": "My 24 Hours in Tokyo",
    "publishedAt": "2025-10-17T03:25:00Z",
    "channelId": "UC1234...",
    "channelTitle": "Traveler A",
    "channelHandle": "@travelerA",
    "categoryId": "22",
    "durationISO": "PT21S",
    "durationSec": 21,
    "lengthBucket": "short"
  },
  "tracking": {
    "status": "complete",
    "discovered_at": "2025-10-17T03:25:00Z",
    "last_polled_at": "2025-10-17T23:20:00Z",
    "poll_count": 78,
    "stop_reason": "age>=24h"
  },
  "stats_snapshots": [
    { "ts": "2025-10-17T03:30:00Z", "viewCount": 1200, "likeCount": 70, "commentCount": 4 },
    { "ts": "2025-10-17T03:45:00Z", "viewCount": 1800, "likeCount": 90, "commentCount": 6 }
  ]
}
```

**Notes:**
- One document per YouTube video (`_id` = videoId).  
- `track_once.py` appends `stats_snapshots` until `tracking.status` becomes `"complete"`.  
- `stop_reason` may be one of: `"age>=24h"`, `"low_quality"`, `"removed"`.

## 2️⃣ Collection: `processed_videos` (CORE)

**Purpose:** Detailed post-processed dataset for analytics and ML model training.  
**Generated by:** `process_data.py` (v7.x) — includes horizon aggregation, feature extraction, and quality labeling.

### Structure
```json
{
  "video_id": "abcd1234",
  "status": "complete",
  "published_at": "2025-10-17T03:25:00Z",
  "n_snapshots": 78,
  "last_snapshot_ts": "2025-10-17T23:20:00Z",
  "completed_horizons": [60, 180, 360, 720, 1440],
  "n_completed_horizons": 5,
  "coverage_score": 0.85,
  "growth_phase": "steady",
  "horizons": {
    "60":   { "views": 1200, "likes": 70, "comments": 4, "value_method": "floor", "coverage_ratio": 0.62, "n_expected": 12, "n_available": 8 },
    "180":  { "views": 4200, "likes": 190, "comments": 21, "value_method": "floor", "coverage_ratio": 0.60, "n_expected": 28, "n_available": 17 },
    "360":  { "views": 9200, "likes": 440, "comments": 48, "value_method": "ceil", "coverage_ratio": 0.51, "n_expected": 44, "n_available": 22 },
    "720":  { "views": 22000, "likes": 900, "comments": 120, "value_method": "floor", "coverage_ratio": 0.80, "n_expected": 52, "n_available": 42 },
    "1440": { "views": 56000, "likes": 2300, "comments": 380, "value_method": "floor", "coverage_ratio": 0.92, "n_expected": 64, "n_available": 59 }
  },
  "snapshot_features": {
    "v_slope_mean": 1.27,
    "v_slope_max": 3.45,
    "v_accel_mean": 0.84,
    "time_first_1k": 90.5,
    "time_first_10k": 540.2
  },
  "ml_flags": {
    "likely_viral": false,
    "viral_confirmed": false,
    "score": 0.21
  },
  "processed_status": "done"
}
```

**Notes:**
- One row per video; aggregates at 1h–3h–6h–12h–24h horizons.  
- `growth_phase ∈ {"early-burst", "steady", "plateau", "flat", "decline"}`.  
- `value_method`: `"floor"` → last snapshot ≤ horizon; `"ceil"` → first ≥ horizon (within 30m).  
- `coverage_score` summarises per-horizon completeness (mean of ratios).

## 3️⃣ Collection: `channels` (SUPPORTING)

**Purpose:** YouTube channel metadata and stats.  
**Populated by:** `discover_once.py` or periodic sync.

### Structure
```json
{
  "_id": "UC1234...",
  "handle": "@travelerA",
  "snippet": {
    "title": "Traveler A",
    "publishedAt": "2018-03-12T00:00:00Z",
    "country": "JP"
  },
  "stats": {
    "subscriberCount": 142000,
    "videoCount": 210,
    "viewCount": 9800000
  },
  "last_synced_at": "2025-10-17T10:00:00Z"
}
```

**Notes:**
- Joins with `videos.snippet.channelId` or `snippet.channelHandle`.  
- Serves as lookup for aggregated dashboard metrics.

## 🧾 Detailed Schemas

Below are **schema-style definitions** for each collection, including field types and enum values. Types are written in a TypeScript-like notation for clarity.

### 1️⃣ `videos` schema

```ts
interface VideoDoc {
  _id: string; // YouTube videoId

  ml_flags?: {
    likely_viral: boolean;
    viral_confirmed: boolean;
    score: number; // 0–1
  };

  source: {
    query: string;
    regionCode: string;      // e.g. "US", "JP"
    randomMode: boolean;
    channelHandle?: string;  // present if discovered via channel
  };

  snippet: {
    title: string;
    publishedAt: string;     // ISO datetime
    channelId: string;
    channelTitle: string;
    channelHandle?: string;
    categoryId: string;      // YouTube category ID
    durationISO: string;     // e.g. "PT21S"
    durationSec: number;
    lengthBucket: "short" | "medium" | "long";
  };

  tracking: {
    status: "tracking" | "complete";
    discovered_at: string;   // ISO
    last_polled_at: string;  // ISO
    poll_count: number;
    stop_reason: (
      | "age>=24h"
      | "low_quality"
      | "removed"
    );
  };

  stats_snapshots: Array<{
    ts: string;          // ISO snapshot time
    viewCount: number;
    likeCount: number | null;
    commentCount: number;
  }>;
}
```

### 2️⃣ `processed_videos` schema

```ts
interface HorizonMetrics {
  views: number;
  likes: number;
  comments: number;
  value_method: "floor" | "ceil";
  coverage_ratio: number; // 0–1
  n_expected: number;
  n_available: number;
}

interface ProcessedVideoDoc {
  _id?: any;              // ObjectId in Mongo (optional in exports)
  video_id: string;       // FK → videos._id

  status: "complete" | "tracking" | "error";

  published_at: string;   // ISO
  n_snapshots: number;
  last_snapshot_ts: string; // ISO

  completed_horizons: number[];         // e.g. [60,180,360,720,1440]
  n_completed_horizons: number;

  coverage_score: number;               // overall coverage 0–1
  growth_phase: "early-burst" | "steady" | "plateau" | "flat" | "decline";

  horizons: {
    "60"?: HorizonMetrics;
    "180"?: HorizonMetrics;
    "360"?: HorizonMetrics;
    "720"?: HorizonMetrics;
    "1440"?: HorizonMetrics;
  };

  snapshot_features?: {
    v_slope_mean?: number;
    v_slope_max?: number;
    v_accel_mean?: number;
    time_first_1k?: number;
    time_first_10k?: number;
  };

  ml_flags?: {
    likely_viral: boolean;
    viral_confirmed: boolean;
    score: number;
  };

  processed_status: "done" | "queued" | "skipped" | "tracking" | "error";
}
```

interface ChannelDoc {
